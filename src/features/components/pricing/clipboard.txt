 * E -> effectivePrice

 * H -> displayCycleTotal
 * P -> isPppActive
 * S -> isOfferApplied
 * R -> strikethroughPrice
 * $ -> priceId
 * q -> calculateSavings
 * L -> showSavings
 * _ -> mappedFeatures
 */

const surfaceClasses = {
  dark: "bg-foreground text-background",
  surface: "bg-surface text-foreground",
  "surface-light": "bg-foreground/[0.06] text-foreground"
};

const totalAnimationCount = pricingUtils.default?.animationStats?.totalCount ?? 50;

const calculateSavings = (plan, pppTier) => {
  if (pppTier) {
    const pppQuarterly = getPppEffectivePrice(plan.key, "quarterly", pppTier);
    const pppYearly = getPppEffectivePrice(plan.key, "yearly", pppTier);
    if (pppQuarterly != null && pppYearly != null) {
      const savings = 4 * pppQuarterly - pppYearly;
      return savings > 0 ? savings : 0;
    }
  }
  const quarterlyEffective = (pricingUtils.effectiveCyclePrice(plan.quarterly) ?? 0) * 12;
  const yearlyEffective = pricingUtils.effectiveCyclePrice(plan.yearly) ?? 0;
  const defaultSavings = quarterlyEffective - yearlyEffective;
  return defaultSavings > 0 ? defaultSavings : 0;
};

export default function PricingCard({
  plan,
  cycle,
  isLoading = false,
  onCheckout,
  rollerDuration = 2,
  latestShip = null,
  shippedRecently = 0
}) {
  const cycleData = plan[cycle];
  const surfaceClass = surfaceClasses[plan.surface] ?? surfaceClasses.surface;
  const isDark = plan.surface === "dark";
  
  const mutedTextClass = isDark ? "text-background/70" : "text-foreground-muted";
  const normalTextClass = isDark ? "text-background/70" : "text-foreground/70";
  
  const offerActive = isOfferActive();
  const pppTier = usePppTier();
  
  const pppDisplayPrice = pppTier ? getPppDisplayPrice(plan.key, cycle, pppTier) : null;
  const effectivePrice = pricingUtils.effectiveCyclePrice(cycleData);
  
  const displayPriceValue = pppDisplayPrice ? pppDisplayPrice.price : effectivePrice;
  const displayCycleTotal = pppDisplayPrice ? pppDisplayPrice.cycleTotal : pricingUtils.effectiveCycleTotal(cycleData);
  
  const isPppActive = !!pppDisplayPrice;
  const isOfferApplied = !isPppActive && offerActive && typeof cycleData.listPrice === "number" && cycleData.listPrice > displayPriceValue;
  const strikethroughPrice = isPppActive ? effectivePrice : isOfferApplied ? cycleData.listPrice : null;
  const priceId = cycleData.priceId;
  
  const savingsAmount = calculateSavings(plan, pppTier);
  const showSavings = cycle === "yearly" && savingsAmount > 0;

  const mappedFeatures = (function(features, { shippedRecently = 0 } = {}) {
    return features.map(feature => {
      if (typeof feature === "string") return feature;
      
      if (feature?.componentCount) {
        return {
          label: t("pricing.tiers.allComponents", { count: totalAnimationCount }),
          highlight: true
        };
      }
      
      if (feature?.shippedRecently) {
        if (shippedRecently >= 2) {
          return {
            label: t("pricing.tiers.shippedRecently", { count: shippedRecently }),
            solid: true
          };
        }
        return null;
      }
      
      return feature?.label || "";
    }).filter(Boolean);
  })(plan.features || [], { shippedRecently }).map(feature => {
    if (typeof feature === "string") return feature;
    
    if (feature.highlight) {
      return {
        label: feature.label,
        className: isDark ? "!text-background font-medium" : "!text-foreground font-medium"
      };
    }
    
    if (feature.solid) {
      return {
        label: feature.label,
        className: isDark ? "!text-background" : "!text-foreground"
      };
    }
    
    return feature;
  });

  return (
    <article
      className={`pricing-card relative flex h-full flex-col p-32 lg:p-40 ${surfaceClass}`}
      data-popular={plan.isPopular ? "true" : undefined}
    >
      <header className="flex items-start justify-between gap-16">
        {plan.isPopular ? (
          <span className="text-mono-sm text-brand">
            {t("pricing.tiers.mostPopular")}
          </span>
        ) : (
          <span aria-hidden="true" />
        )}
        
        {plan.seatTag ? (
          <span className={`text-mono-sm ${normalTextClass}`}>
            {plan.seatTag}
          </span>
        ) : null}
      </header>

      <div className="mt-32">
        <p className={`text-h4 font-normal ${normalTextClass}`}>
          {plan.name}
        </p>

        {isPppActive ? (
          <div className="mt-4 flex justify-start">
            <PppLabel />
          </div>
        ) : isOfferApplied ? (
          <p className="mt-4 text-mono-sm text-brand">
            {t("pricing.tiers.offerName")}
          </p>
        ) : null}

        <div
          className="mt-8 flex items-baseline gap-6"
          aria-label={
            isPppActive
              ? t("pricing.tiers.pppAria", { price: effectivePrice, cycle: cycleData.cycleLabel })
              : isOfferApplied
              ? t("pricing.tiers.offerAria", { price: cycleData.listPrice, cycle: cycleData.cycleLabel })
              : undefined
          }
        >
          <span className="text-h2 font-medium tracking-tight inline-flex items-baseline">
            <span>€</span>
            <AnimatedNumber
              value={displayPriceValue}
              triggerMode="immediate"
              duration={rollerDuration}
              valueChangeDuration={0.8}
              className="text-h2 font-medium leading-none"
            />
          </span>

          {strikethroughPrice != null ? (
            <span className={`text-body line-through ${mutedTextClass}`}>
              €{strikethroughPrice}
            </span>
          ) : null}

          <span className={`text-body ${mutedTextClass}`}>
            {cycleData.cycleLabel}
          </span>
        </div>

        {displayCycleTotal ? (
          <p className={`text-body-sm mt-6 ${mutedTextClass}`}>
            {isPppActive
              ? t("pricing.tiers.billedQuarterlyExact", { total: displayCycleTotal })
              : t("pricing.tiers.billedQuarterly")}
          </p>
        ) : null}

        <div
          className="grid transition-all duration-500 ease-power4-in-out"
          style={{ gridTemplateRows: showSavings ? "1fr" : "0fr" }}
          aria-hidden={!showSavings}
        >
          <div className="overflow-hidden">
            <p className="text-mono-sm pt-12 text-brand">
              {t("pricing.tiers.savings", { amount: savingsAmount })}
            </p>
          </div>
        </div>
      </div>

      {plan.pitch ? (
        <p className={`text-body mt-24 max-w-[28ch] ${mutedTextClass}`}>
          {plan.pitch}
        </p>
      ) : null}

      <div className="mb-auto mt-32">
        <FeatureList
          items={mappedFeatures}
          itemClassName={`text-body leading-snug ${isDark ? "text-background/90" : "text-foreground/85"}`}
        />
        
        {latestShip ? (
          <p className={`text-body-sm mt-20 ${mutedTextClass}`}>
            {t("pricing.tiers.latestShip")}:{" "}
            <Link
              href={`/animations/${latestShip.slug}`}
              inline={true}
              className={isDark ? "text-background" : "text-foreground"}
            >
              {latestShip.title}
            </Link>{" "}
            ·{" "}
            {relativeShipped(latestShip.published_at || latestShip.created_at)}
          </p>
        ) : null}
      </div>

      <div className="mt-40 flex">
        <Button
          theme="brand"
          size="sm"
          className="w-full justify-center sm:w-auto"
          onClick={() => {
            analytics.track("pricing_plan_selected", {
              plan: plan.key,
              cycle: cycle,
              price: displayPriceValue,
              ...(pppTier ? { ppp_tier: pppTier } : {})
            });
            onCheckout?.(plan.key, priceId);
          }}
          aria-label={t("pricing.tiers.ctaAria", { cta: plan.cta, cycle })}
          loading={isLoading}
        >
          {plan.cta}
        </Button>
      </div>

      <p className={`text-body-sm mt-16 ${mutedTextClass}`}>
        {t("pricing.tiers.cardFinePrint")}
      </p>
    </article>
  );
}
